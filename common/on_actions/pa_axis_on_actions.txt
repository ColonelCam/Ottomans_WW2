   if = {
    limit = {

     country_exists = GER
     country_exists = POL
     country_exists = UKR
     GER = {
      is_subject = no

      NOT = { has_war_with = POL }

     }
     POL = { NOT = { has_war_with = UKR } }

     if = {

      limit = { NOT = { is_in_faction_with = POL } }

   }

 }
