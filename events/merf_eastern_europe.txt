	desc = merf_r.6.d

	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		UKR = {
			country_event = {
				id = merf_r.7
				days = 7
			}
		}
	}
	trigger = {
		OR = {
			country_exists = POL
			country_exists = UKR
		}
	}
	#keep the war going
	option = {
		name = merf_r.6.a
		remove_ideas = SOV_great_polish_revolt
		trigger = {
			POL = {
				is_in_faction = yes
			}
		}
		POL = {
			set_cosmetic_tag = POL_big_poland
		}
	}
	option = {
		name = merf_r.6.a
		remove_ideas = SOV_great_polish_revolt
		trigger = {
			POL = {
				is_in_faction = no
			}
		}
		POL = {
			set_cosmetic_tag = POL_big_poland
		}
		if = {
			limit = {
				country_exists = POL
			}
		every_owned_state = {
			limit = {
				is_core_of = POL
			}
			set_state_owner_to = POL
		}
		}
		if = {
			limit = {
				country_exists = UKR
			}
		every_owned_state = {
			limit = {
				is_core_of = UKR
			}
			set_state_owner_to = UKR
		}
		}
		white_peace = POL
		white_peace = UKR
	}
}
